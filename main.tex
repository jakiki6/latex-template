\section{Analyse}
\subsection{AIDA}
Attention, Interest, Desire, Action

\subsection{Visuelle Ebenen der Kamera}
\begin{itemize}
    \item Supertotale: alles zu sehen, fuer umfassenden Ueberblick -> Kontext und raeumliche Anordnung von Charaktern und Umgebung
    \item Totale: wichtigste Elemente zu sehen + Umgebung, Charaktere und Ereignisse
    \item Halbtotale: Hauptfigur + Teil der Umgegung -> Fokus auf Charaktere oder bestimme Handlungselemente
    \item Amerikanische: ungefaehr auf Huefthoehe oder leicht drunter -> Fokus auf Charaktere mit Gesicht und Oberkoerper, oft in Westernfilmen fuer Duelle
    \item Halbnahe: Huefte oder Brust aufwaerts -> Mimik, Gestik, Reaktionen + immernoch Umgebung zu sehen
    \item Nahe: bestimmtes Element (z.B. Gesicht) -> starket Fokus auf Emotionen, Reaktionen oder Details
    \item Grosse: Gesicht oder bestimmtes Detail -> zum Aufmerksamkeitslenken + emotionale Ausdruecke
    \item Supergrosse: Teil des Gesichts -> emotionale Ausdruecke oder subtile Nuancen
    \item Detail: sehr kleines Detail -> besondere Aufmerksamkeit
    \item Italienische: nur Augen
\end{itemize}

\subsection{Perspektive}
\begin{itemize}
    \item Froschperspektive: vom Boden
    \item Untersicht: leicht von unten -> macht Charakter groesser
    \item Vogelperspektive: von ganz oben -> Ueberblick
    \item POV: stellt Sicht des Charakters dar
    \item Aufsicht: leicht von oben -> macht Charakter kleiner
    \item Neutral: Augenhoehe -> vermittel gleichheit
    \item Overshoulder: beide Personen im Bild -> vermittelt Verbundenheit
\end{itemize}

\subsection{Farben}
\begin{itemize}
    \item warme Farben wie Rot: Leidenschaft
    \item kalte Farben wie Blau: ruhige Atmosphaere
    \item koennen symbolisch sein, um Themen oder Charaktere zu repraesentieren
\end{itemize}

\subsection{Bildkomposition}
\begin{itemize}
    \item Kontraste: ruhig/unruhig, alt/neu, ...
    \item Fluessige Bewegungen: kontrollierte, gezielte Bewegungen, Effekte wie Parallax
    \item Drittel Regel oder Goldener Schnitt
    \item Fill the Frame: nah rangehen, um Bildschirm zu fuellen -> naeher = relevanter
    \item Vordergrunde: Hinter- und Vordergruende fuer Dimensionen
    \item Symetrien: professionelle und aesthetische Wirkung
    \item Framing: gerade Haltung der Kamera, natuerliche Rahmen wie z.B. Baeume
    \item Dominanz: stoerende Elemente ausblenden
    \item Goldene/Blaue Stunde
    \item Fuehungslienien: spannend, leiten Blick, natuerliche und kuenstliche Linien
\end{itemize}

\subsection{Erzaehlebenen}
\begin{itemize}
    \item Visuell: was sehe ich?
    \item Auditiv: was hoere ich?
    \item narrativ: was passiert?
\end{itemize}
